    <script>
        let selectedPackageData = null;

        function fetchSpotifyData() {
            const input = document.getElementById('spotifyLinkInput').value;
            if(!input.trim()) {
                alert('Por favor ingresa un enlace válido de Spotify.');
                return;
            }
            
            // Simulación visual de lectura de metadatos de Spotify para que el cliente identifique su tema
            const preview = document.getElementById('spotifyPreview');
            const img = document.getElementById('previewImg');
            const title = document.getElementById('previewTitle');
            const artist = document.getElementById('previewArtist');
            const type = document.getElementById('previewType');

            preview.classList.remove('hidden');
            img.src = "https://images.unsplash.com/photo-1511671782779-c97d3d27a1d4?w=150&auto=format&fit=crop&q=80";
            title.innerText = "Look At U (White Noise EP)";
            artist.innerText = "MAIK THE HEIGHT / Casa Music";
            type.innerText = input.includes('playlist') ? 'PLAYLIST' : 'TRACK';
            
            alert('¡Pista detectada y vinculada correctamente con la base de datos interna!');
        }

        function selectPackage(name, price, days) {
            selectedPackageData = { name, price, days };
            document.getElementById('summarySection').classList.remove('hidden');
            document.getElementById('sumPackage').innerText = name;
            document.getElementById('sumDays').innerText = days + " días";
            document.getElementById('sumPrice').innerText = "$" + price + " USD";
            
            // Scroll suave hacia el resumen
            document.getElementById('summarySection').scrollIntoView({ behavior: 'smooth' });
        }

        function simulateCheckout() {
            alert('¡Pago procesado con éxito vía Stripe! El sistema ha creado el Smart Link interno y ha disparado la API de Meta Ads para iniciar la pauta publicitaria de forma automatizada.');
            location.reload();
        }
    </script>
